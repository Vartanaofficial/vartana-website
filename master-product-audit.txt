function getDirectImageUrl(rawUrl: string): string {
  const url = rawUrl.trim();
  if (!url) return "";

  const driveFileMatch = url.match(
    /drive\.google\.com\/file\/d\/([a-zA-Z0-9_-]+)/
  );

  const driveOpenMatch = url.match(
    /drive\.google\.com\/(?:open|uc)\?.*?[?&]id=([a-zA-Z0-9_-]+)/
  );

  const fileId = driveFileMatch?.[1] || driveOpenMatch?.[1];

  if (fileId) {
    return `https://drive.google.com/uc?export=view&id=${fileId}`;
  }

  return url;
}

import { NextRequest, NextResponse } from "next/server";
import { verifyAdminSessionToken } from "@/lib/admin-auth";

export const runtime = "nodejs";
export const dynamic = "force-dynamic";

const TIMEOUT_MS = 30000;

function isAdmin(request: NextRequest): boolean {
  const token = request.cookies.get("vartana_admin_session")?.value;
  const secret = process.env.ADMIN_SESSION_SECRET;

  return Boolean(
    token &&
    secret &&
    secret.length >= 32 &&
    verifyAdminSessionToken(token, secret)
  );
}

function getScriptUrl(): URL | null {
  const raw = process.env.NEXT_PUBLIC_GOOGLE_SCRIPT_URL;
  if (!raw) return null;

  try {
    const url = new URL(raw);
    if (url.protocol !== "https:" || url.hostname !== "script.google.com") {
      return null;
    }
    return url;
  } catch {
    return null;
  }
}

async function scriptRequest(url: URL, init?: RequestInit) {
  const controller = new AbortController();
  const timeout = setTimeout(() => controller.abort(), TIMEOUT_MS);

  try {
    const response = await fetch(url.toString(), {
      ...init,
      cache: "no-store",
      signal: controller.signal,
    });

    const text = await response.text();

    if (!response.ok) {
      throw new Error(`Apps Script HTTP ${response.status}`);
    }

    let data: Record<string, unknown>;
    try {
      data = JSON.parse(text);
    } catch {
      throw new Error("Apps Script memberikan respons bukan JSON.");
    }

    return data;
  } finally {
    clearTimeout(timeout);
  }
}

export async function GET(request: NextRequest) {
  if (!isAdmin(request)) {
    return NextResponse.json(
      { success: false, message: "Sesi admin tidak valid." },
      { status: 401 }
    );
  }

  const url = getScriptUrl();
  if (!url) {
    return NextResponse.json(
      { success: false, message: "Konfigurasi Apps Script tidak tersedia." },
      { status: 500 }
    );
  }

  url.searchParams.set("action", "getProducts");

  try {
    const data = await scriptRequest(url);
    return NextResponse.json({
      success: true,
      products: Array.isArray(data.products) ? data.products : [],
    });
  } catch (error) {
    console.error("Admin products GET failed:", error);
    return NextResponse.json(
      { success: false, message: "Gagal mengambil master produk." },
      { status: 502 }
    );
  }
}

export async function POST(request: NextRequest) {
  if (!isAdmin(request)) {
    return NextResponse.json(
      { success: false, message: "Sesi admin tidak valid." },
      { status: 401 }
    );
  }

  const url = getScriptUrl();
  if (!url) {
    return NextResponse.json(
      { success: false, message: "Konfigurasi Apps Script tidak tersedia." },
      { status: 500 }
    );
  }

  let body: Record<string, unknown>;

  try {
    body = await request.json();
  } catch {
    return NextResponse.json(
      { success: false, message: "Format data tidak valid." },
      { status: 400 }
    );
  }

  const sku = String(body.sku ?? "").trim();
  const productName = String(body.productName ?? "").trim();

  if (!/^[A-Za-z0-9._-]{2,80}$/.test(sku)) {
    return NextResponse.json(
      { success: false, message: "Format SKU tidak valid." },
      { status: 400 }
    );
  }

  if (!productName || productName.length > 200) {
    return NextResponse.json(
      { success: false, message: "Nama produk wajib diisi (maksimal 200 karakter)." },
      { status: 400 }
    );
  }

  const optionalTextFields = [
    "productNumber",
    "variantName",
    "subtitle",
    "status",
    "description",
    "imageUrl",
  ] as const;

  const product: Record<string, unknown> = {
    sku,
    productName,
    active: body.active !== false,
  };

  for (const field of optionalTextFields) {
    if (body[field] !== undefined) {
      if (typeof body[field] !== "string") {
        return NextResponse.json(
          { success: false, message: `Field ${field} harus berupa teks.` },
          { status: 400 }
        );
      }
      product[field] = (body[field] as string).trim();
    }
  }

  if (
    product.imageUrl &&
    (typeof product.imageUrl !== "string" ||
      !/^https:\/\//i.test(product.imageUrl))
  ) {
    return NextResponse.json(
      { success: false, message: "URL gambar harus menggunakan HTTPS." },
      { status: 400 }
    );
  }

  for (const field of ["hpp", "weightGram"] as const) {
    if (body[field] !== undefined) {
      const value = Number(body[field]);
      if (!Number.isFinite(value) || value < 0) {
        return NextResponse.json(
          { success: false, message: `${field} harus berupa angka nol atau lebih.` },
          { status: 400 }
        );
      }
      product[field] = value;
    }
  }

  try {
    // Tentukan create/update dari SKU yang sudah tersimpan.
    const readUrl = new URL(url);
    readUrl.searchParams.set("action", "getProducts");
    const existingData = await scriptRequest(readUrl);
    const existingProducts = Array.isArray(existingData.products)
      ? existingData.products as Array<Record<string, unknown>>
      : [];

    const exists = existingProducts.some(
      (item) =>
        String(item.sku ?? "").trim().toUpperCase() === sku.toUpperCase()
    );

    const saveUrl = new URL(url);
    const payload = {
      action: exists ? "updateProduct" : "createProduct",
      ...product,
    };

    const result = await scriptRequest(saveUrl, {
      method: "POST",
      headers: { "Content-Type": "text/plain;charset=utf-8" },
      body: JSON.stringify(payload),
    });

    if (result.success !== true) {
      return NextResponse.json(
        {
          success: false,
          message: String(result.message ?? "Apps Script menolak penyimpanan."),
        },
        { status: 502 }
      );
    }

    return NextResponse.json({
      success: true,
